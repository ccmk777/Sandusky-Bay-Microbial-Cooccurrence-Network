subset_phyloseq_Site1 <- aggregate_taxa(merged_phyloseq_data, "Genus")

#Get the OTU names for nodes (if we skip this the names will include all previous taxa levels)
taxic <- as.data.frame(merged_phyloseq_data@tax_table)
Network_Names <- taxic$Genus


# Get the abundance data (OTU table) from the phyloseq object
abundance_data_site1 <- otu_table(t(merged_phyloseq_data))

# Calculate the Spearman correlation matrix
cor_matrix <- cor(abundance_data_site1, method = "spearman")

# Set edge weights to zero for correlations that don't meet the absolute value threshold
threshold <- 0.6
cor_matrix[cor_matrix >= -threshold & cor_matrix <= threshold] <- 0

# Create an igraph graph object from the binary adjacency matrix
cooccurrence_network <- graph_from_adjacency_matrix(cor_matrix, mode = "undirected", weighted = TRUE, diag = FALSE)

# Add the OTU names (node labels) to the network
V(cooccurrence_network)$name <- Network_Names

# Get the edge weights (correlation values)
edge_weights <- E(cooccurrence_network)$weight

# Choose a layout algorithm (layout_nicely)
layout_algorithm <- "nicely"

# Define colors for positive and negative edge weights
positive_color <- "blue"
negative_color <- "red"

# Visualize the network using ggraph
Merged18.16.Network.Genus <- ggraph(cooccurrence_network, layout = layout_algorithm) + 
  geom_edge_link(aes(edge_width = abs(edge_weights), color = ifelse(edge_weights >= threshold, positive_color, negative_color)),
                 edge_alpha = 0.6) +
  scale_edge_color_identity() +
  geom_node_point() +
  geom_node_text(aes(label = name), repel = TRUE) +
  theme_void()

Merged18.16.Network.Genus
